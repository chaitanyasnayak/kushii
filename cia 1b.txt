//Create Customer Table
CREATE TABLE Customer (
    Customer_ID INT PRIMARY KEY,
    Name VARCHAR(30) NOT NULL,
    Mobile_No VARCHAR(10) UNIQUE,
    Age INT CHECK (Age >= 18)
);


//Create Recharge Table
CREATE TABLE Recharge (
    Recharge_ID INT PRIMARY KEY,
    Customer_ID INT,
    Amount DECIMAL(10,2) CHECK (Amount > 0),
    Recharge_Date DATE,
    Recharge_Time TIME,
    FOREIGN KEY (Customer_ID) REFERENCES Customer(Customer_ID)
);

//inner join
SELECT Customer.Customer_Name,
       Recharge.Recharge_ID,
       Recharge.Amount
FROM Customer
INNER JOIN Recharge
ON Customer.Customer_ID = Recharge.Customer_ID;


//left join
SELECT Customer.Customer_Name,
       Recharge.Recharge_ID,
       Recharge.Amount
FROM Customer
LEFT JOIN Recharge
ON Customer.Customer_ID = Recharge.Customer_ID;


//right join
SELECT Customer.Customer_Name,
       Recharge.Recharge_ID,
       Recharge.Amount
FROM Customer
RIGHT JOIN Recharge
ON Customer.Customer_ID = Recharge.Customer_ID;


//fullouter join

       Recharge.Recharge_ID,
       Recharge.Amount
FROM Customer
LEFT JOIN Recharge
ON Customer.Customer_ID = Recharge.Customer_ID

//self join
SELECT A.Customer_Name AS Customer1,
       B.Customer_Name AS Customer2
FROM Mobile_Recharge A
JOIN Mobile_Recharge B
ON A.Customer_ID < B.Customer_ID;

/sub query
SELECT Customer_Name, Amount
FROM Mobile_Recharge
WHERE Amount > (
    SELECT AVG(Amount)
    FROM Mobile_Recharge
);

//correlated 
SELECT A.Customer_Name, A.Amount
FROM Mobile_Recharge A
WHERE A.Amount > (
    SELECT B.Amount
    FROM Mobile_Recharge B
    WHERE B.Recharge_ID = A.Recharge_ID - 1
);
