DDL – Data Definition Language

//Create Customer table
CREATE TABLE Customer (
    Customer_ID INT PRIMARY KEY,
    Name VARCHAR(30),
    Mobile_No VARCHAR(15),
    Balance DECIMAL(10,2)
);

//Create Recharge table
CREATE TABLE Recharge (
    Recharge_ID INT PRIMARY KEY,
    Customer_ID INT,
    Amount DECIMAL(10,2),
    Recharge_Date DATE,
    FOREIGN KEY (Customer_ID) REFERENCES Customer(Customer_ID)
);


//Add a column
ALTER TABLE Customer ADD Email VARCHAR(50);
ALTER TABLE Customer MODIFY Name VARCHAR(50);

//Create ServiceProvider table
CREATE TABLE ServiceProvider (
    Provider_ID INT PRIMARY KEY,
    Provider_Name VARCHAR(30)
);

GRANT SELECT, UPDATE
ON MobileRecharge.Customer

ON MobileRecharge.Customer
FROM 'recharge_user'@'localhost';
