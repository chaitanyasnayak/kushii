    <p id="op"></p>
    <input type="submit" value="Show" onclick="show()">
    <script>
        async function show(){
            const response = await fetch('https://fakestoreapi.com/products');
            const rp = await response.json();
            let res = "";
            for (let i = 0; i < 5; i++) {
                res += "ID : " + rp[i].id + "<br>" + "Title : " + rp[i].title + "<br>" + "Price : " + rp[i].price + "<br>" + "Category : " + rp[i].category + "<br>" + "Rating : " + rp[i].rating + "<br><br>";
            }
            document.getElementById("op").innerHTML = res;
        }
    </script>
