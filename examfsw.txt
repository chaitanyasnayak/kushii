Exercise 12: MongoDB CRUD Operations with Node.js Driver   [MongoDB & Node.js]
const { MongoClient } = require("mongodb");
 
async function run() {
  const c = new MongoClient("mongodb://127.0.0.1:27017");
  await c.connect();
  let s = c.db("MCU").collection("students");
 
  await s.insertOne({name:"Priya"});
  await s.updateOne({name:"Priya"},{$set:{name:"Madhu"}});
  await s.deleteOne({name:"Madhu"});
  console.log(await s.find().toArray());
  await c.close();
}
run();
