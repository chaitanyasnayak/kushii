function check() {
    var f = document.getElementById("fname").value;

    var d = document.getElementById("dob").value;
    var p = document.getElementById("phone").value;
    var e = document.getElementById("email").value;

    if(f=="" || l=="" || d=="" || p=="" || e=="") {
        alert("All fields required");
        return false;
    }

    if(!e.includes("@")) {
        alert("Invalid Email");
        return false;
    }

    if(p.length != 10 || isNaN(p)) {
        alert("Invalid Phone");
        return false;
    }

    alert("Registration Successful");
    return true;
}

    <script>
        // 1. Setup Dates on Page Load using a shortcut
        const todayStr = new Date().toISOString().split('T')[0];
        
        let eighteenAgo = new Date();
        eighteenAgo.setFullYear(eighteenAgo.getFullYear() - 18);
        const maxDobStr = eighteenAgo.toISOString().split('T')[0];

        // Restrict calendar pickers
        document.getElementById('todayDate').setAttribute('min', todayStr);
        document.getElementById('todayDate').setAttribute('max', todayStr);
        document.getElementById('dob').setAttribute('max', maxDobStr);

        // 2. Validate on Submit
        document.getElementById('registerForm').onsubmit = function(event) {
            let isValid = true;

            // SHORTCUT: A single function handles showing/hiding every error
            function check(errorId, condition) {
                document.getElementById(errorId).style.display = condition ? 'none' : 'block';
                if (!condition) isValid = false;
            }

            // Get values easily
            const val = (id) => document.getElementById(id).value.trim();

            // Run Checks
            check('nameError', val('fullName') !== '');
            check('todayDateError', val('todayDate') === todayStr);
            
            // Check DOB is 18+
            let dob = val('dob');
            let age = new Date().getFullYear() - new Date(dob).getFullYear();
            check('dobError', dob !== '' && age >= 18);
            
            check('emailError', val('email').includes('@'));
            
            // Check Phone is 10 digits and only numbers
            let phone = val('phone');
            check('phoneError', phone.length === 10 && !isNaN(phone));
            
            check('genderError', document.querySelector('input[name="gender"]:checked') !== null);
            check('photoError', document.getElementById('photo').files.length > 0);
            check('passwordError', val('password').length >= 8);

            // Stop form submission if there are errors
            if (!isValid) {
                event.preventDefault();
            } else {
                event.preventDefault(); // Remove this line in a real project
                alert("Form submitted successfully! You are 18+.");
            }
        };
    </script>
