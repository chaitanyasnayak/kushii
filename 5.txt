USE nutri;


CREATE TABLE Users (
    user_id INT PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    age INT CHECK(age >= 18),
    email VARCHAR(50) UNIQUE
);


CREATE TABLE Food (
    food_id INT PRIMARY KEY,
    food_name VARCHAR(50) NOT NULL,
    calories INT CHECK(calories > 0),
    user_id INT FOREIGN KEY REFERENCES Users(user_id)
);




CREATE TABLE MealPlan (
    meal_id INT PRIMARY KEY,
    meal_name VARCHAR(50) NOT NULL,
    meal_type VARCHAR(30),
    user_id INT FOREIGN KEY REFERENCES Users(user_id)
);


CREATE TABLE Exercise (
    exercise_id INT PRIMARY KEY,
    exercise_name VARCHAR(50) NOT NULL,
    duration INT CHECK(duration > 0),
    calories_burned INT CHECK(calories_burned > 0),
    user_id INT FOREIGN KEY REFERENCES Users(user_id)
);

-- 1c(e). Joins

SELECT Users.name, Food.food_name
FROM Users
INNER JOIN Food
ON Users.user_id = Food.user_id;


SELECT Users.name, Food.food_name
FROM Users
LEFT JOIN Food
ON Users.user_id = Food.user_id;


SELECT Users.name, Food.food_name
FROM Users
RIGHT JOIN Food
ON Users.user_id = Food.user_id;


SELECT Users.name, Food.food_name
FROM Users
FULL JOIN Food
ON Users.user_id = Food.user_id;


-- Join with MealPlan

-- Join with Exercise

-- 1c(f). Subqueries

SELECT name, age
FROM Users
WHERE age > (SELECT AVG(age) FROM Users);


SELECT food_name, calories
FROM Food
WHERE calories > (SELECT AVG(calories) FROM Food);


-- Correlated Subquery

SELECT name
FROM Users U
WHERE EXISTS (
    SELECT *
    FROM Food F
    WHERE F.user_id = U.user_id
);

-- 2b(b). T-SQL Blocks

-- Fibonacci

DECLARE @n INT = 10;
DECLARE @a INT = 0;
DECLARE @b INT = 1;
DECLARE @c INT;

WHILE @a <= @n
BEGIN
    PRINT @a;

    SET @c = @a + @b;
    SET @a = @b;
    SET @b = @c;
END;


-- Sum of First N Natural Numbers

DECLARE @n INT = 10;
DECLARE @i INT = 1;
DECLARE @sum INT = 0;

WHILE @i <= @n
BEGIN
    SET @sum = @sum + @i;
    SET @i = @i + 1;
END;

PRINT @sum;
